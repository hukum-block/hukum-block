class MülakatModal(discord.ui.Modal, title="Mülakat Soruları"):
    q1 = discord.ui.TextInput(label="Eyo Paşa Roblox İsmi Nedir?", style=discord.TextStyle.short, required=True)
    q2 = discord.ui.TextInput(label="Örnek aldığı kişiler kimler? (ve ile ayır)", style=discord.TextStyle.short, required=True)
    q3 = discord.ui.TextInput(label="Şifre nedir? (Araya boşluk bırakarak yaz)", style=discord.TextStyle.short, required=True)

    async def on_submit(self, interaction: discord.Interaction):
        await interaction.response.defer(ephemeral=True) # Zaman aşımını önlemek için ilk adım
        roblox_cevap = self.q1.value.strip().lower()
        ornek_cevap = self.q2.value.strip().upper()
        sifre_cevap = self.q3.value.strip()

        if roblox_cevap == "mr_eyo98" and ("ATATÜRK" in ornek_cevap and "ENVER" in ornek_cevap) and sifre_cevap == "1881-1922 1881-1938":
            member = interaction.user
            verified_role = discord.utils.get(interaction.guild.roles, name="Doğrulanmış")
            unverified_role = discord.utils.get(interaction.guild.roles, name="doğrulanmadı")

            if verified_role:
                await member.add_roles(verified_role)
            if unverified_role:
                await member.remove_roles(unverified_role)

            await interaction.followup.send("Tebrikler mülakatı geçtin! Sistem doğrulaması tamamlandı. Gizli kamp odaları emrinizzedir, paşam!", ephemeral=True)

async def installation(interaction: discord.Interaction):
    await interaction.response.defer(ephemeral=True)
    guild = interaction.guild

    try:
        verified_role = discord.utils.get(guild.roles, name="Doğrulanmış")
        if not verified_role:
            verified_role = await guild.create_role(name="Doğrulanmış", color=discord.Color.green())

        unverified_role = discord.utils.get(guild.roles, name="doğrulanmadı")
        if not unverified_role:
            unverified_role = await guild.create_role(name="doğrulanmadı", color=discord.Color.red())

        register_channel = discord.utils.get(guild.text_channels, name="kayıt")
        if not register_channel:
            overwrites = {
                guild.default_role: discord.PermissionOverwrite(view_channel=True, send_messages=False)
            }
            register_channel = await guild.create_text_channel(name="kayıt", overwrites=overwrites)

        gizli_kanallar = ['osmanlı-kampı-özel', 'modern-kamp-özel', 'polis-kampı-özel', 'kamp-rütbeleri-ss', 'özel-bilgiler']
        
        for kanal_adi in gizli_kanallar:
            check_chan = discord.utils.get(guild.text_channels, name=kanal_adi)
            if not check_chan:
                overwrites = {
                    guild.default_role: discord.PermissionOverwrite(view_channel=False),
                    verified_role: discord.PermissionOverwrite(view_channel=True)
                }
                await guild.create_text_channel(name=kanal_adi, overwrites=overwrites)

        embed = discord.Embed(
            title="Eyo Paşa Doğrulama Sistemi",
            description="Aşağıdaki butona basarak soruları doğru cevapla ve gizli kammpalara erişim sağla.",
            color=discord.Color.blue()
        )
        
        await register_channel.send(embed=embed, view=VerifyView())
        await interaction.followup.send("Sistem ve gizli odalar başarıyla kuruldu, paşam!", ephemeral=True)

    except Exception as e:
        print(e)
        await interaction.followup.send("Kurulum sırasında bir hata çıktı.", ephemeral=True)

def get_member_rank_tier(member, kamp):
    for role in member.roles:

        if "HC" in role.name: return 3
        if "HR" in role.name: return 2
        if "MR" in role.name: return 1
    return 0

    @discord.ui.button(label="📩 Destek Talebi Aç", style=discord.ButtonStyle.danger, custom_id="bilet_ac_butonu")
    async def bilet_ac(self, interaction: discord.Interaction, button: discord.ui.Button):
        await interaction.response.defer(ephemeral=True) # Zaman aşımını önlemek için ilk adım
        guild = interaction.guild
        kategori = discord.utils.get(guild.categories, name="--- BİLETLER / DESTEK ---")
        if not kategori:
            kategori = await guild.create_category(name="--- BİLETLER / DESTEK ---")

        mevcut_kanal = discord.utils.get(guild.text_channels, name=bilet_kanali_adi)

            await interaction.followup.send(f"🚨 Zaten açık bir talep odanız var paşam: {mevcut_kanal.mention}", ephemeral=True)
            return

        overwrites = {
            guild.default_role: discord.PermissionOverwrite(view_channel=False),
            interaction.user: discord.PermissionOverwrite(view_channel=True, send_messages=True, attach_files=True),
        }
        
        for role in guild.roles:
            if role.permissions.administrator:
                overwrites[role] = discord.PermissionOverwrite(view_channel=True, send_messages=True)

        bilet_kanali = await guild.create_text_channel(name=bilet_kanali_adi, category=kategori, overwrites=overwrites)
        
        embed = discord.Embed(
            title="⚔️ HÜKÜM DESTEK MERKEZİ ⚔️",
            description=f"Selamün Aleyküm {interaction.user.mention} paşam. Destek talebiniz açılmıştır. Yetkililerimiz birazdan burada hazır bulunacaktır. İntizamı bozmadan sorununuzu yazabilirsiniz.",
            color=discord.Color.red()
        )
        # Bilet kapatma butonunu buraya ekliyoruz
        await bilet_kanali.send(embed=embed, view=BiletKapatView())
        await interaction.followup.send(f"✅ Talebiniz oluşturuldu paşam: {bilet_kanali.mention}", ephemeral=True)
